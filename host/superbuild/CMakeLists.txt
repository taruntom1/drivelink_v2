cmake_minimum_required(VERSION 3.24)
include(${CMAKE_CURRENT_LIST_DIR}/../../cmake/DriveLinkVersion.cmake)
project(drivelink_host VERSION ${DRIVELINK_VERSION} LANGUAGES CXX)
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 13.1)
  message(FATAL_ERROR "DriveLink host requires GCC 13.1 or newer for C++23")
endif()
if(WIN32 AND (NOT MINGW OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU" OR NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL 13.1.0))
  message(FATAL_ERROR "Windows host builds require Qt's MinGW GCC 13.1.0 kit")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT WIN32 AND NOT DEFINED CMAKE_CXX_COMPILER_LAUNCHER AND NOT "$ENV{CCACHE_DISABLE}" STREQUAL "1")
  find_program(_drivelink_ccache ccache)
  if(_drivelink_ccache)
    set(CMAKE_CXX_COMPILER_LAUNCHER "${_drivelink_ccache}")
  endif()
endif()
include(CTest)
option(DRIVELINK_BUILD_GUI "Build drivelink-gui (needs Qt 6.10 or newer)" ON)
if(WIN32)
  set(_drivelink_agent_default OFF)
else()
  set(_drivelink_agent_default ON)
endif()
option(DRIVELINK_BUILD_AGENT "Build drivelink-agent (Linux only)" ${_drivelink_agent_default})
find_package(Threads REQUIRED)
include(../cmake/Dependencies.cmake)
include(../cmake/Json.cmake)
add_compile_options("$<$<CXX_COMPILER_ID:GNU,Clang>:-Wall;-Wextra;-Werror>")
add_subdirectory(../../protocol protocol)
add_subdirectory(../libdrivelink libdrivelink)
add_subdirectory(../vdev vdev)
add_subdirectory(../cli cli)
if(DRIVELINK_BUILD_AGENT)
  include(GNUInstallDirs)
  add_subdirectory(../agent agent)
endif()
if(DRIVELINK_BUILD_GUI)
  add_subdirectory(../gui gui)
endif()
if(BUILD_TESTING)
  add_subdirectory(../test test)
endif()
