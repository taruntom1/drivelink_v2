cmake_minimum_required(VERSION 3.24)
project(drivelink_host VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
find_package(Threads REQUIRED)
include(../cmake/Dependencies.cmake)
include(../cmake/Json.cmake)
add_compile_options("$<$<CXX_COMPILER_ID:GNU,Clang>:-Wall;-Wextra;-Werror>")
add_subdirectory(../../protocol protocol)
add_subdirectory(../libdrivelink libdrivelink)
add_subdirectory(../vdev vdev)
add_subdirectory(../cli cli)
if(BUILD_TESTING)
  add_subdirectory(../test test)
endif()
