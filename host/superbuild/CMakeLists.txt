cmake_minimum_required(VERSION 3.24)
project(drivelink_host VERSION 0.1.0 LANGUAGES CXX)
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 13.1)
  message(FATAL_ERROR "DriveLink host requires GCC 13.1 or newer for C++23")
endif()
if(WIN32 AND (NOT MINGW OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU" OR NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL 13.1.0))
  message(FATAL_ERROR "Windows host builds require Qt's MinGW GCC 13.1.0 kit")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
find_package(Threads REQUIRED)
include(../cmake/Dependencies.cmake)
include(../cmake/Json.cmake)
add_compile_options("$<$<CXX_COMPILER_ID:GNU,Clang>:-Wall;-Wextra;-Werror>")
add_subdirectory(../../protocol protocol)
add_subdirectory(../libdrivelink libdrivelink)
add_subdirectory(../vdev vdev)
add_subdirectory(../cli cli)
if(BUILD_TESTING)
  add_subdirectory(../test test)
endif()
