cmake_minimum_required(VERSION 3.24)
include(${CMAKE_CURRENT_LIST_DIR}/../../cmake/DriveLinkVersion.cmake)
project(drivelink VERSION ${DRIVELINK_VERSION} LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
find_package(Threads REQUIRED)
if(NOT TARGET drivelink::protocol)
  find_package(drivelink_protocol REQUIRED)
endif()
include(../cmake/Dependencies.cmake)
add_library(drivelink STATIC src/session.cpp src/client.cpp src/values.cpp src/model.cpp src/groups.cpp src/buffers.cpp src/command.cpp src/capture.cpp src/testgen.cpp src/time.cpp src/tuning.cpp src/ota.cpp src/ota_update.cpp src/link.cpp src/ports.cpp src/agent_wire.cpp src/discovery.cpp src/discovery_net.cpp)
add_library(drivelink::drivelink ALIAS drivelink)
target_compile_features(drivelink PUBLIC cxx_std_23)
# The ROS plugin (host/ros/drivelink_hardware) links this static library into a shared object.
set_target_properties(drivelink PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(drivelink PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(drivelink PUBLIC drivelink::protocol PRIVATE $<BUILD_INTERFACE:drivelink_asio> $<BUILD_INTERFACE:drivelink_mdns> Threads::Threads)
if(WIN32)
  target_link_libraries(drivelink PRIVATE ws2_32 mswsock setupapi advapi32 iphlpapi)
endif()
install(TARGETS drivelink EXPORT drivelinkTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT drivelinkTargets NAMESPACE drivelink:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/drivelink)
configure_package_config_file(cmake/drivelinkConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/drivelinkConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/drivelink)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/drivelinkConfigVersion.cmake VERSION ${DRIVELINK_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/drivelinkConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/drivelinkConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/drivelink)
